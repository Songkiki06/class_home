<script>
    // ⚠️ 본인의 구글 웹 앱 URL 주소 링크를 정확히 채워주세요.
    const GAS_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbyr8oBznbtYk7TZ9wmJeQ9lMbF-6A8ExKc3RA0n8zKcom7iaJKQ20fVJthWymVtS-I/exec";

    // 학교 공식 6종 타임슬록 셋업
    const timeSlots = [
        { name: "아침 자습 (08:00 - 09:00)", max: 1 },
        { name: "오전 일과 (09:00 - 13:00)", max: 4 }, 
        { name: "오후 일과 (14:00 - 18:00)", max: 4 }, 
        { name: "저녁 자습 (19:00 - 22:00)", max: 3 }, 
        { name: "야간 자습 (22:00 - 00:00)", max: 2 }, 
        { name: "심야 공부 (00:00 - 01:00)", max: 1 }
    ];

    const tbody = document.getElementById('planner-tbody');
    timeSlots.forEach((slot, index) => {
        let options = `<option value="선택 안함">선택 안함</option>`;
        for (let h = 0.5; h <= slot.max; h += 0.5) {
            let displayLabel = "";
            if (h === 0.5) {
                displayLabel = "30분";
            } else if (Number.isInteger(h)) {
                displayLabel = `${h}시간`;
            } else {
                displayLabel = `${Math.floor(h)}시간 30분`;
            }
            options += `<option value="${displayLabel}">${displayLabel}</option>`;
        }

        tbody.innerHTML += `
            <tr class="time-row">
                <td class="time-col">${slot.name}</td>
                <td class="hour-col">
                    <select class="hour-select">
                        ${options}
                    </select>
                </td>
                <td class="subject-col"><input type="text" class="sub-input" placeholder="과목"></td>
                <td class="content-col"><input type="text" class="con-input" placeholder="핵심 내용 요약"></td>
            </tr>
        `;
    });

    function updatePlaceholder() {
        const selectedType = document.querySelector('input[name="info-type"]:checked').value;
        document.getElementById('student-input').placeholder = `${selectedType}을(를) 입력해주세요.`;
        document.getElementById('student-input').value = "";
        document.getElementById('planner-content').style.display = 'none';
        document.getElementById('status-msg').style.display = 'none';
    }

    function checkStudent() {
        const type = document.querySelector('input[name="info-type"]:checked').value;
        const value = document.getElementById('student-input').value.trim();
        const statusMsg = document.getElementById('status-msg');
        const authBtn = document.getElementById('auth-btn');
        
        if(!value) { alert(`${type}을 입력하세요.`); return; }
        
        statusMsg.style.display = 'block';
        statusMsg.style.color = '#555';
        statusMsg.textContent = '구글 시트에서 명단 조회 중...';
        authBtn.disabled = true;

        fetch(`${GAS_WEB_APP_URL}?action=check&type=${encodeURIComponent(type)}&value=${encodeURIComponent(value)}`)
            .then(res => res.json())
            .then(result => {
                authBtn.disabled = false;
                if(result.success) {
                    statusMsg.style.color = 'var(--success-color)';
                    statusMsg.textContent = `✓ 인증 성공: [학번: ${result.studentId} / 이름: ${result.name}] 학생 접속 확인`;
                    document.getElementById('planner-content').style.display = 'block';
                } else {
                    statusMsg.style.color = 'var(--danger-color)';
                    statusMsg.textContent = `✗ 인증 실패: 명단 시트에 정보가 존재하지 않습니다.`;
                    document.getElementById('planner-content').style.display = 'none';
                }
            })
            .catch(err => {
                authBtn.disabled = false;
                alert('명단 서버 통신에 실패했습니다. 웹 앱 주소를 확인해 주세요.');
            });
    }

    function submitPlanner() {
        const type = document.querySelector('input[name="info-type"]:checked').value;
        const value = document.getElementById('student-input').value.trim();
        const reviewVal = document.getElementById('review-input').value.trim();
        const saveBtn = document.getElementById('final-save-btn');
        
        var plannerData = [];
        document.querySelectorAll('.time-row').forEach(row => {
            var hourVal = row.querySelector('.hour-select').value;
            var subVal = row.querySelector('.sub-input').value.trim();
            var conVal = row.querySelector('.con-input').value.trim();
            
            if(subVal || conVal || hourVal !== "선택 안함") {
                plannerData.push({
                    timeBlock: row.querySelector('.time-col').textContent, // 레이블 그대로 추출
                    studyTime: hourVal,
                    subject: subVal,
                    content: conVal
                });
            }
        });

        if(plannerData.length === 0 && !reviewVal) { alert('기재된 내용이 없습니다.'); return; }

        saveBtn.disabled = true;
        saveBtn.textContent = '구글 시트에 전송 중...';

        const payload = {
            studentInfo: {type: type, value: value},
            plannerData: plannerData, // 구조화된 원본 데이터를 그대로 배열 전송
            reviewData: reviewVal
        };

        fetch(GAS_WEB_APP_URL, {
            method: "POST",
            mode: "no-cors", 
            headers: { "Content-Type": "text/plain;charset=utf-8" }, 
            body: JSON.stringify(payload)
        }).then(() => {
            alert('구글 시트에 성공적으로 저장되었습니다!');
            saveBtn.disabled = false;
            saveBtn.textContent = '구글 시트에 플래너 저장하기';
            clearInputs();
        }).catch(err => {
            alert('전송 중 에러가 발생했습니다.');
            saveBtn.disabled = false;
            saveBtn.textContent = '구글 시트에 플래너 저장하기';
        });
    }

    function clearInputs() {
        document.querySelectorAll('.sub-input, .con-input').forEach(i => i.value = "");
        document.querySelectorAll('.hour-select').forEach(s => s.value = "선택 안함");
        document.getElementById('review-input').value = "";
    }
</script>
